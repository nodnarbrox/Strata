cmake_minimum_required(VERSION 3.24)
project(strata_liquid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(strata-liquid SHARED engine.cpp)
target_include_directories(strata-liquid PRIVATE "${STRATA_ROOT}/include")

# Explicit Linux linkage to an existing matching GGML header/library bundle.
# No inference server or llama.cpp engine is loaded by this path.
set(STRATA_LIQUID_GGML_INCLUDE_DIR "" CACHE PATH "Matching external GGML headers")
set(STRATA_LIQUID_GGML_CPU_LIBRARY "" CACHE FILEPATH "External GGML CPU shared library")
set(STRATA_LIQUID_GGML_BASE_LIBRARY "" CACHE FILEPATH "External GGML base shared library")
if(STRATA_LIQUID_GGML_INCLUDE_DIR OR STRATA_LIQUID_GGML_CPU_LIBRARY OR STRATA_LIQUID_GGML_BASE_LIBRARY)
  if(NOT STRATA_LIQUID_GGML_INCLUDE_DIR OR NOT STRATA_LIQUID_GGML_CPU_LIBRARY OR NOT STRATA_LIQUID_GGML_BASE_LIBRARY)
    message(FATAL_ERROR "External Liquid GGML requires all three STRATA_LIQUID_GGML_* paths")
  endif()
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "External Liquid GGML linkage is currently supported on Linux only")
  endif()
  foreach(_header ggml.h ggml-cpu.h)
    if(NOT EXISTS "${STRATA_LIQUID_GGML_INCLUDE_DIR}/${_header}")
      message(FATAL_ERROR "Missing external GGML header: ${_header}")
    endif()
  endforeach()
  foreach(_library STRATA_LIQUID_GGML_CPU_LIBRARY STRATA_LIQUID_GGML_BASE_LIBRARY)
    if(NOT EXISTS "${${_library}}" OR IS_DIRECTORY "${${_library}}")
      message(FATAL_ERROR "Missing external GGML shared library: ${${_library}}")
    endif()
  endforeach()
  add_library(strata-liquid-external-cpu SHARED IMPORTED)
  add_library(strata-liquid-external-base SHARED IMPORTED)
  set_target_properties(strata-liquid-external-cpu PROPERTIES IMPORTED_LOCATION "${STRATA_LIQUID_GGML_CPU_LIBRARY}")
  set_target_properties(strata-liquid-external-base PROPERTIES IMPORTED_LOCATION "${STRATA_LIQUID_GGML_BASE_LIBRARY}")
  target_include_directories(strata-liquid PRIVATE "${STRATA_LIQUID_GGML_INCLUDE_DIR}")
  target_link_libraries(strata-liquid PRIVATE strata-liquid-external-cpu strata-liquid-external-base)
  target_compile_definitions(strata-liquid PRIVATE STRATA_LIQUID_REFERENCE_KV_PADDING=1)
else()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_CUDA OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE ON CACHE BOOL "" FORCE)
  add_subdirectory("${STRATA_ROOT}/third_party/strata-ggml/ggml" "${CMAKE_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
  target_link_libraries(strata-liquid PRIVATE ggml-cpu ggml-base)
endif()
if(NOT WIN32)
  target_link_options(strata-liquid PRIVATE -Wl,-Bsymbolic)
endif()
